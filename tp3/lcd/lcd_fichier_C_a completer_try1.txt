void ecriture_commande(char instruction) {
	RS = 0;  // Set RS to 0 for command mode
	RW = 0;  // Set RW to 0 for write mode
 	EN = 0;  // Set EN to 0 initially

    // Send high nibble (4 most significant bits) of the instruction
    LCD_PORT = (LCD_PORT & 0x0F) | (instruction & 0xF0);  // Mask and set high nibble   //trouver le register the register that controls the data lines 												//connected to the LCD
    EN = 1;   // Enable LCD
    __delay_ms(1);  // Short delay
    EN = 0;   // Disable LCD

    // Send low nibble (4 least significant bits) of the instruction
    LCD_PORT = (LCD_PORT & 0x0F) | ((instruction << 4) & 0xF0);  // Shift and set low nibble
    EN = 1;   // Enable LCD
    __delay_ms(1);  // Short delay
    EN = 0;   // Disable LCD

    // Wait for the command execution time
    if ((instruction == 0x01) || (instruction == 0x02) || (instruction == 0x03) || (instruction == 0x04) || (instruction == 0x10) || (instruction == 0x11) || (instruction == 0x12) || (instruction == 0x13)) // je comprend pas trop mais bon
        __delay_ms(2);  // Longer delay for specific commands
    else
        __delay_ms(40); // Short delay for other commands
}

void print_char(char character) {
    RS = 1;  // Set RS to 1 for data mode
    RW = 0;  // Set RW to 0 for write mode
    EN = 0;  // Set EN to 0 initially

    // Send high nibble (4 most significant bits) of the character
    LCD_PORT = (LCD_PORT & 0x0F) | (character & 0xF0);  // Mask and set high nibble
    EN = 1;   // Enable LCD
    __delay_ms(1);  // Short delay
    EN = 0;   // Disable LCD

    // Send low nibble (4 least significant bits) of the character
    LCD_PORT = (LCD_PORT & 0x0F) | ((character << 4) & 0xF0);  // Shift and set low nibble
    EN = 1;   // Enable LCD
    __delay_ms(1);  // Short delay
    EN = 0;   // Disable LCD

    // Wait for the character execution time
    __delay_ms(40); // Short delay
}
